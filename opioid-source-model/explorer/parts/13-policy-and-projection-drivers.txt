Projection start time[Proj]= INITIAL(
	MIN(Projection last data time[Proj],MaxDataTime))
	~	year
	~	Note: MaxDataTime functions as override for projections data. Only \
		exogenous data prior to MaxDataTime will be used in generating \
		projections. If MaxDataTime is later than last available data point, then \
		exogenous data up to the last available data point will be used and \
		projection curves will be fitted to the last available data point.

Projection last data value[Proj]  = 
        INITIAL( GET DATA AT TIME ( Projection input data[Proj] , 
                  Projection start time[Proj] )  )
	~	
	~

Projection output data[Proj]  = 
        IF THEN ELSE ( Time 
                  <= Projection start time[Proj] , 
             Projection input data[Proj] , 
             Switch for constant projections[Proj] 
                  * Projection input data[Proj] 
                  + ( 1
                       - Switch for constant projections[Proj] ) 
                       * Logistic growth curve[Proj] )
	~	
	~	Trend projections to extend the input (exogenous) time series data used to \
		drive the base model.

Logistic growth curve[Proj]  = 
        IF THEN ELSE ( Projection curve end value[Proj] 
                  > Projection last data value[Proj] , 
             MIN ( Projection curve end value[Proj] , 
                  ( ( Projection curve end value[Proj] 
                       - Projection last data value[Proj] ) 
                       / 0.5) 
                       / ( 1
                            + EXP ( ( -5
                                      / ( Projection curve end time[Proj] 
                                           - Projection start time[Proj] ) ) 
                                      * ( Time 
                                           - Projection start time[Proj] ) ) ) 
                       + ( Projection curve end value[Proj] 
                            - ( Projection curve end value[Proj] 
                                 - Projection last data value[Proj] ) 
                                 / 0.5) ) , 
             Max ( Projection curve end value[Proj] , 
                  ( ( Projection curve end value[Proj] 
                       - Projection last data value[Proj] ) 
                       / 0.5) 
                       / ( 1
                            + EXP ( ( -5
                                      / ( Projection curve end time[Proj] 
                                           - Projection start time[Proj] ) ) 
                                      * ( Time 
                                           - Projection start time[Proj] ) ) ) 
                       + ( Projection curve end value[Proj] 
                            - ( Projection curve end value[Proj] 
                                 - Projection last data value[Proj] ) 
                                 / 0.5) ) )
	~	
	~

	Average MME per day relative 
	             ^ (4 * Sensitivity of Rx supply to MME per day / (Sensitivity of Rx supply to MME per day\
		 + Sensitivity of Rx supply to patients receiving prescription
	 + Sensitivity of Rx supply to Rx per person + Sensitivity of Rx supply to days per prescription\
		))
	             * Patients receiving opioid prescription annual relative 
	                  ^ (4 * Sensitivity of Rx supply to patients receiving prescription\
		/ (Sensitivity of Rx supply to MME per day
	 + Sensitivity of Rx supply to patients receiving prescription + Sensitivity of Rx supply to Rx per person\
		 + Sensitivity of Rx supply to MME per day)) * Prescriptions per person relative
	 ^ (4 * Sensitivity of Rx supply to Rx per person / (Sensitivity of Rx supply to MME per day\
		 + Sensitivity of Rx supply to patients receiving prescription
	 + Sensitivity of Rx supply to Rx per person + Sensitivity of Rx supply to days per prescription\
		)) 
	* Average days per prescription relative
	             ^ (4 * Sensitivity of Rx supply to MME per day / (Sensitivity of Rx supply to MME per day\
		 + Sensitivity of Rx supply to patients receiving prescription
	 + Sensitivity of Rx supply to Rx per person + Sensitivity of Rx supply to days per prescription\
		))
	~	dmnl
	~	Supply of prescription opioids available for potential misuse, indexed to \
		model start. Can be thought of as 'excess' pills not used as prescribed \
		within the time period of the prescription, which therefore present \
		potential opportunities for misuse.
